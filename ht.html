this.charts = {}; }, methods: { async fetchChartData(type) { let url = ""; switch (type) { case "requests": url = "/api/service_requests/stats"; break; case "trends": url = "/api/service_requests/monthly"; break; case "topServices": url = "/api/services/popular"; break; case "ratings": url = "/api/professionals/ratings"; break; case "pincode": url = "/api/service_requests/pincode_distribution"; break; } try { const res = await fetch(url); const data = await res.json(); if (res.ok) { this.renderChart(type, data); } else { console.error("Failed to fetch chart data:", data.message); } } catch (error) { console.error("Error fetching chart data:", error); } }, renderChart(type, data) { const chartId = `${type}Chart`; const chartElement = document.getElementById(chartId); if (!chartElement) { console.error(`Canvas element with ID '${chartId}' not found.`); return; } // Check and destroy the existing chart instance, if any if (this.charts[type]) { try { this.charts[type].destroy(); } catch (err) { console.warn(`Error destroying chart: ${err.message}`); } this.charts[type] = null; // Clear the reference } // Reset canvas element for a clean state const parent = chartElement.parentNode; const newCanvas = document.createElement("canvas"); newCanvas.id = chartId; parent.removeChild(chartElement); // Remove the old canvas parent.appendChild(newCanvas); // Add a fresh canvas const ctx = newCanvas.getContext("2d"); // Define the chart configuration based on the type let config = {}; if (type === "requests") { config = { type: "pie", data: { labels: ["Pending", "Accepted", "Completed", "Rejected"], datasets: [ { data: data.counts, backgroundColor: ["blue", "green", "orange", "red"], }, ], }, }; } else if (type === "trends") { config = { type: "line", data: { labels: data.months, datasets: [ { label: "Requests", data: data.counts, borderColor: "blue" }, ], }, }; } else if (type === "topServices") { config = { type: "bar", data: { labels: data.services, datasets: [ { label: "Bookings", data: data.counts, backgroundColor: "purple", }, ], }, }; } else if (type === "ratings") { config = { type: "bar", data: { labels: data.professionals, datasets: [ { label: "Ratings", data: data.ratings, backgroundColor: "gold" }, ], }, }; } else if (type === "pincode") { config = { type: "bar", data: { labels: data.pincodes, datasets: [ { label: "Requests", data: data.counts, backgroundColor: "teal" }, ], }, }; } // Create a new chart instance try { this.charts[type] = new Chart(ctx, config); } catch (err) { console.error(`Error creating chart: ${err.message}`); } }, }, };
